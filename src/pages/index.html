{{de}}Linux-Pakete aus der Paketquelle apt.mirranet.de (Debian 12+, Ubuntu 22.04+). APT prüft jedes Paket mit dem Signaturschlüssel der Quelle; Updates kommen danach mit sudo apt upgrade:{{en}}Linux packages from the package repository apt.mirranet.de (Debian 12+, Ubuntu 22.04+). APT checks every package against the repository's signing key; updates then arrive with sudo apt upgrade:{{/}}

sudo install -d /etc/apt/keyrings
sudo curl -fsSL https://apt.mirranet.de/mirranet.gpg -o /etc/apt/keyrings/mirranet.gpg
echo "deb [signed-by=/etc/apt/keyrings/mirranet.gpg] https://apt.mirranet.de stable main" \
    | sudo tee /etc/apt/sources.list.d/mirranet.list
sudo apt update
